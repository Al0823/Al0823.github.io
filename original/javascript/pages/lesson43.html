The if Statement

let age = 18;

if (age >= 18) {
  console.log("You are an adult.");
}

The else Statement

let age = 16;

if (age >= 18) {
  console.log("Adult");
} else {
  console.log("Minor");
}

The else if Statement

let score = 85;

if (score >= 90) {
  console.log("Grade A");
} else if (score >= 80) {
  console.log("Grade B");
} else {
  console.log("Keep trying!");
}
